namespace tida_inf04czerwiec25;

public class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("Podaj tekst do zaszyfrowania:");
        string tekst = Console.ReadLine() ?? string.Empty;
        Console.WriteLine("Podaj klucz szyfru cezara:");
        int klucz = int.Parse(Console.ReadLine() ?? "0");

        SzyfrCezara szyfrCezara = new SzyfrCezara(tekst, klucz);
        Console.WriteLine(szyfrCezara.Szyfruj());
    }
}

public class SzyfrCezara
{
    private string alfabet = "abcdefghijklmnopqrstuvwxyz";
    private string tekst;
    private int klucz;

    public SzyfrCezara(string tekst, int klucz)
    {
        this.tekst = tekst;
        this.klucz = klucz;
    }

    public string Szyfruj()
    {
        if (klucz == 0)
        {
            return tekst;
        }

        string zaszyfrowanyNapis = "";
        int przesuniecie = ((klucz % alfabet.Length) + alfabet.Length) % alfabet.Length;
        foreach (var litera in tekst)
        {
            int indeks = alfabet.IndexOf(litera);
            if (indeks < 0)
            {
                zaszyfrowanyNapis += litera;
                continue;
            }

            int nowyIndeks = (indeks + przesuniecie) % alfabet.Length;
            char zaszyfrowana = alfabet[nowyIndeks];
            zaszyfrowanyNapis += zaszyfrowana;
        }

        return zaszyfrowanyNapis;
    }
}
